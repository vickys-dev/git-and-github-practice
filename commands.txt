
# to setup git 
git config --global user.name "your name"
git config --global user.email "youremail@gmail.com"

# to initializ git repo
git init 

# to check the status
git status

# to add what to commit
git add "file name"
git add . // the . means to tell git add all files created, modified and deleted

# to commit changes
git commit -m 'added readme.md' // -m means message

# to see all the commit histery
git log

# to check the previous version or switch commit version with the commit id
git checkout commit-id-here
git checkout master  or main // to swithch to latest commit
git checkout -f master or main // -f to force to switch to latest commit

# to switch to main branch
git branch -M main

# to link the remote repo to our local repo
git remote add origin https://github.com/vickys-dev/git-and-github-practice.git

# to update our local commits to our remote repo
git push -u origin main
