
# to setup git 
git config --global user.name "your name"
git config --global user.email "youremail@gmail.com"

# to initializ git repo
git init 

# to check the status
git status

# to add what to commit
git add "file name"
git add . // the . means to tell git add all files created, modified and deleted

# to commit changes
git commit -m 'added readme.md' // -m means message

# to see all the commit histery
git log

# to check the previous version or switch commit version with the commit id
git checkout commit-id-here
git checkout master  or main // to swithch to latest commit
git checkout -f master or main // -f to force to switch to latest commit

# to switch to main branch
git branch -M main

# to link the remote repo to our local repo
git remote add origin https://github.com/vickys-dev/git-and-github-practice.git

# to update our local commits to our remote repo
git push -u origin main

# to create new branch 
git branch branch-name
git checkout branch-name // to switch to branch with branch name
git checkout main // to switch to main branch
git checkout -b feature-branch // to create and switch to branch
git branch new-branch-name source-branch // the branch will create based on the branch that we currently are. So, we need to make sure the new branch name with the source branch name
git push --set-upstream origin feature-branch // git push -u origin feature-branch in this command -u is short for --set-upstream
git push // this will update our local branch to remote branch after linking it with --set-upstream
git pull // this will pull the changes others make in our remote branch


# the workfolw that we follow most of the time
- Clone the repo
- Create a new branch from the main or another branch
- Make your changes
- Push the branch to the remote repo
- Open up a pull request
- Merge the changes
- Pull the merged changes into your local main branch
- Repeat form step 2
